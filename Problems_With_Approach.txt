public int findKthLargest(int[] nums, int k) {
    Queue<Integer> minHeap = new PriorityQueue<>();

    for (int i = 0; i < nums.length; i++) {
        minHeap.add(nums[i]);

        if (minHeap.size() > k) {
            minHeap.poll(); // removes the head (smallest) and rebalances the tree
        }
    }

    return minHeap.peek();
}

S.C: O(k)

53. Kadane's Algorithm:

It is used to solve problems with finding the maximum subarray sum (positives + negatives)

Approach:

1. Initialize current_sum to first element or 0. Initialize max_sum to first element or minimum integer value.

2. Iterate over the array and add the element to current_sum.

3. Find the maximum between current_sum and max_sum and update the value of max_sum.

4. If the value of current_sum goes below 0, then we set current_sum to 0 because if the value is negative then if we add the next value then its sum is going to be less than the added value.

public int maxSubArray(int[] nums) {
	int max_sum = Integer.MIN_VALUE;
	int current_sum = 0;

	for (int i = 0; i < nums.length; i++) {
		current_sum += nums[i];

		max_sum = Math.max(max_sum, current_sum);

		if (current_sum < 0) {
			current_sum = 0;
		}
	}

	return max_sum;
}

T.C: O(n), S.C: O(1)

To rotate an n x n 2D matrix by 90 degrees (clockwise)

[4,5,6]	 ->	 [8,5,2]
[7,8,9]		 [9,6,3]

[4,5,6]

2. Take transpose of each element
Transpose of matrix[i][j] = matrix[j][i]

public void rotate(int[][] matrix) {
	// Flip the matrix upside down
	int rowLength = matrix[0].length;
	int colLength = matrix.length;
	int k = colLength - 1;

	for (int i = 0; i < (colLength / 2); i++) {
		for (int j = 0; j < rowLength; j++) {
			int temp = matrix[i][j];
			matrix[i][j] = matrix[k][j];
			matrix[k][j] = temp;
		}
		k--;
	}

	// Transpose of matrix
	for (int i = 0; i < colLength; i++) {
		for (int j = i + 1; j < rowLength; j++) { // Transpose only the upper right half to avoid double transpose
			int temp = matrix[i][j];
			matrix[i][j] = matrix[j][i];
			matrix[j][i] = temp;
		}
	}
}

15. 3Sum

public List<List<Integer>> threeSum(int[] nums) {
	List<List<Integer>> result = new ArrayList<List<Integer>>();
	int n = nums.length;

	// We'll be using the 2 pointer approach to find the second and third element. 2 pointer approach works on an sorted array.
	Arrays.sort(nums);

	for (int i = 0; i < n; i++) {
		// If the previous element is same as the current element then there is a chance that the result array would be the same
		// For e.g. [-1, -1, 0, 1, 2] will have [-1, -1, 2], [-1, 0, 1] and [-1, 0, 1] where one array is duplicate
		if (i > 0 && nums[i] == nums[i - 1]) {
			continue;
		}

		int start = i + 1;
		int end = n - 1;

		while (start < end) {
			int total = nums[i] + nums[start] + nums[end];

			if (total > 0) {
				end--;
			} else if (total < 0) {
				start++;
			} else {
				result.add(Arrays.asList(nums[i], nums[start], nums[end]));
				start++;
				end--;

				// We have to change both the elements to avoid duplicates.
				// For e.g. [0, 0, 0, 0] = [i, start, 0, end] and after start++ [i, 0, start, end]. i, start and end is same in both cases
				while (start < end && nums[start] == nums[start - 1]) {
					start++;
				}
				
				while (start < end && nums[end] == nums[end + 1]) {
                    end--;
                }
			}
		}
	}

	return result;
}

18. 4Sum
public List<List<Integer>> fourSum(int[] nums, int target) {
	List<List<Integer>> result = new ArrayList<List<Integer>>();
	int n = nums.length;

	Arrays.sort(nums);

	for (int i = 0; i < n; i++) {
		if (i > 0 && nums[i] == nums[i - 1]) {
			continue;
		}

		for (int j = i + 1; j < n; j++) {
			if (j != i + 1 && nums[j] == nums[j - 1]) {
				continue;
			}

			int start = j + 1;
			int end = n - 1;

			while (start < end) {
				long sum = (long) nums[i] + nums[j] + nums[start] + nums[end];

				if (sum > target) {
					end--;
				} else if (sum < target) {
					start++;
				} else {
					result.add(Arrays.asList(nums[i], nums[j], nums[start], nums[end]));
					start++;
					end--;

					while (start < end && nums[start] == nums[start - 1]) {
						start++;
					}

					while (start < end && nums[end] == nums[end + 1]) {
						end--;
					}
				}
			}
		}
	}

	return result;
}

1838. Frequency of the most frequent element

public int maxFrequency(int[] nums, int k) {
	int n = nums.length;
	int left = 0, right = 0;
	long sum = 0;
	int max = Integer.MIN_VALUE;
	Arrays.sort(nums);

	while (right < n) {
		sum += nums[right];

		while ((long) nums[right] * (right - left + 1) > (sum + k)) {
			sum -= nums[left];
			left++;
		}

		max = Math.max(max, right - left + 1);
		right++;
	}

	return max;

}

public int lengthOfLongestSubstring(String s) {
	int maxLength = 0;
	int left = 0;
	Set<Character> charSet = new HashSet<>();

	for (int i = 0; i < s.length(); i++) {
		if (charSet.contains(s.charAt(i))) {
			while (charSet.contains(s.charAt(i))) {
				charSet.remove(s.charAt(left));
				left++;
			}
		}
		
		maxLength = Math.max(maxLength, i - left + 1);

		charSet.add(s.charAt(i));
	}

	return maxLength;
}

46. Permutations

public List<List<Integer>> permute(int[] nums) {
	return getPermutations(nums, 0, nums.length, new ArrayList<List<Integer>>());
}

private List<List<Integer>> getPermutations(int[] nums, int idx, int n, List<List<Integer>> result) {
	// Base case
	if (idx == n) {
		result.add(new ArrayList<Integer>(Arrays.stream(nums).boxed().collect(Collectors.toList())));
		return result;
	}

	for (int i = idx; i < n; i++) {
		// Swap
		int t = nums[idx];
		nums[idx] = nums[i];
		nums[i] = t;

		getPermutations(nums, idx + 1, n, result);

		// Swap back to original
		t = nums[idx];
		nums[idx] = nums[i];
		nums[i] = t;
	}

	return result;
}

704. Binary Search
public int search(int[] nums, int target) {
	int n = nums.length;
	int low = 0;
	int high = n - 1;

	while (low <= high) {
		int mid = low + (high - low) / 2;

		if (nums[mid] == target) {
			return mid;
		}

		if (target < nums[mid]) {
			high = mid - 1;
		} else {
			low = mid + 1;
		}
	}

	return -1;
}

11. Container with most water

Approach (Sliding window):
1. Initialize left pointer to first element and right to the last element.
2. Calculate the distance between them and multiply by the smallest pillar.
3. Calculate the maximum.
4. If the left pillar is small, move the left pointer, else move the right pointer.

public int maxArea(int[] height) {
	int n = height.length;
	int left = 0;
	int right = n - 1;
	int max = Integer.MIN_VALUE;

	while (left < right) {
		int minPillar = Math.min(height[left], height[right]);
		max = Math.max(max, (right - left) * minPillar);

		if (height[left] < height[right]) {
			left++;
		} else {
			right--;
		}
	}

	return max;
}

public ListNode getIntersectionNode(ListNode headA, ListNode headB) {
	ListNode a = headA;
	ListNode b = headB;

	while (a != b) {
		a = a == null ? headB : a.next;
		b = b == null ? headA : b.next;
	}

56. Merge Intervals
public int[][] merge(int[][] intervals) {

	// 1. Sort the array by the start time
	// 2. Declare two variables (prevStart and prevEnd) which will hold the min and max time of the sub-interval
	// 3. Iterate the intervals:
	// a. If the start time of current interval is less than the end time of previous interval
	// that means its an overlapping interval. Store the max and min time of this interval in prevStart
	// and prevEnd.
	// b. Else add the interval (prevStart and prevEnd) to the result and reassign prevStart and prevEnd to the current interval
	// Remember to add prevStart and prevEnd to the result at the end of the loop to avoid missing the last interval
	
	if (intervals.length <= 1) return intervals;

	Arrays.sort(intervals, (a, b) -> Integer.compare(a[0], b[0]));
	List<List<Integer>> result = new ArrayList<>();
	int prevStart = intervals[0][0];
	int prevEnd = intervals[0][1];

	for (int i = 0; i < intervals.length; i++) {
		int start = intervals[i][0];
		int end = intervals[i][1];

		if (start <= prevEnd) {
			prevEnd = Math.max(prevEnd, end);
			prevStart = Math.min(prevStart, start);
		} else {
			result.add(Arrays.asList(prevStart, prevEnd));
			prevStart = start;
			prevEnd = end;
		}
	}

	result.add(Arrays.asList(prevStart, prevEnd));

	// Convert list to array
	int[][] resultArray = new int[result.size()][2];

	for (int i = 0; i < result.size(); i++) {
		resultArray[i][0] = result.get(i).get(0);
		resultArray[i][1] = result.get(i).get(1);
	}

	return resultArray;

}

57. Insert Interval

public int[][] insert(int[][] intervals, int[] newInterval) {
	int i = 0;
	int n = intervals.length;
	List<int[]> mergedIntervals = new ArrayList<>();

	// While end time of current interval is less than the start time of new interval, add it to result because it is non over-lapping.
	while (i < n && intervals[i][1] < newInterval[0]) {
		mergedIntervals.add(intervals[i]);
		i++;
	}

	// While the start of current interval is less than or equal to the end of new interval, it is overlapping. And since it is overlapping, we find the min and max of our new overlapped interval.
	while (i < n && intervals[i][0] <= newInterval[1]) {
		newInterval[0] = Math.min(newInterval[0], intervals[i][0]);
		newInterval[1] = Math.max(newInterval[1], intervals[i][1]);
		i++;
	}

	mergedIntervals.add(newInterval);

	// Add the remaining intervals
	while (i < n) {
		mergedIntervals.add(intervals[i]);
		i++;
	}

	int[][] result = new int[mergedIntervals.size()][2];

	for (int j = 0; j < mergedIntervals.size(); j++) {
		result[j] = mergedIntervals.get(j);
	}

	return result;

}

class MyLinkedList {

    private class Node {
        int val;
        Node next;

        public Node(int val) {
            this.val = val;
            this.next = null;
        }
    }

    Node head = null;
    Node tail = null;
    int currentSize = 0;

    public MyLinkedList() {
        
    }
    
    public int get(int index) {
        if (index >= currentSize) {
            return -1;
        }

        Node curr = head;

        for (int i = 0; i < index; i++) {
            curr = curr.next;
        }

        return curr.val;
    }
    
    public void addAtHead(int val) {
        Node newNode = new Node(val);
        newNode.next = head;
        head = newNode;

        if (currentSize == 0) {
            tail = head;
        }

        currentSize++;
    }
    
    public void addAtTail(int val) {
        Node newNode = new Node(val);

        if (currentSize == 0) {
            tail = newNode;
            head = tail;
        } else {
            tail.next = newNode;
            tail = newNode;
        }

        currentSize++;
    }
    
    public void addAtIndex(int index, int val) {
        if (index > currentSize) {
            return;
        } else if (index == 0) {
            addAtHead(val);
        } else if (index == currentSize) {
            addAtTail(val);
        } else {
            Node newNode = new Node(val);
            Node prevNode = getNodeAtIndex(index - 1);
            newNode.next = prevNode.next;
            prevNode.next = newNode;
            currentSize++;
        }
    }
    
    public void deleteAtIndex(int index) {
        if (index < 0 || index >= currentSize) {
            return;
        } else if (currentSize == 1) {
            head = null;
            tail = null;
        } else if (index == 0) {
            head = head.next;
        } else if (index == currentSize - 1) {
            Node prevNode = getNodeAtIndex(index - 1);
            prevNode.next = null;
            tail = prevNode;
        } else {
            Node prevNode = getNodeAtIndex(index - 1);
            prevNode.next = prevNode.next.next;
        }

        currentSize--;
    }

    private Node getNodeAtIndex(int index) {
        if (index < 0 || index >= currentSize) {
            return null;
        }

        Node node = head;

        for (int i = 0; i < index; i++) {
            node = node.next;
        }

        return node;
    }
}

146. LRU Cache
class LRUCache {
    int capacity;
    Map<Integer, Node> cache;
    // dummy head and tail to remove the LRU and add the resently used node at the begining
    Node head = new Node(-1, -1);
    Node tail = new Node(-1, -1);

    public LRUCache(int capacity) {
        this.capacity = capacity;
        cache = new HashMap<>();
        head.next = tail;
        tail.prev = head;
    }

    private void delete(Node node) {
        Node next = node.next;
        Node prev = node.prev;
        prev.next = next;
        next.prev = prev;
    }

    private void insert(Node node) {
        Node next = head.next;
        head.next = node;
        node.prev = head;
        node.next = next;
        next.prev = node;
    }
    
    public int get(int key) {
        if (cache.containsKey(key)) {
            Node node = cache.get(key);
            delete(node);
            insert(node);
            cache.put(key, node);

            return node.value;
        }

        return -1;
    }
    
    public void put(int key, int value) {
        if (cache.containsKey(key)) {
            Node node = cache.get(key);
            cache.remove(key);
            delete(node);
        }

        if (capacity == cache.size()) {
            cache.remove(tail.prev.key);
            delete(tail.prev);
        }

        insert(new Node(key, value));
        cache.put(key, head.next);
    }
}

class Node {
    int key;
    int value;
    Node next;
    Node prev;

    public Node(int key, int value) {
        this.key = key;
        this.value = value;
        this.next = null;
        this.prev = null;
    }
}

public int minSubArrayLen(int target, int[] nums) {
        int n = nums.length;
        int left = 0, right = 0, sum = 0;
        int minLength = Integer.MAX_VALUE;

        while (left < n && right < n) {
            sum += nums[right];

            while (sum >= target) {
                minLength = Math.min(minLength, right - left + 1);
                sum -= nums[left];
                left++;
            }
            
            right++;
        }

        if (minLength == Integer.MAX_VALUE) {
            minLength = 0;
        }

        return minLength;
    }

496. Next Greater Element I

public int[] nextGreaterElement(int[] nums1, int[] nums2) {
	int[] result = new int[nums1.length];
	Stack<Integer> stack = new Stack<>();
	Map<Integer, Integer> nextGreater = new HashMap<>();

	for (int i = nums2.length - 1; i >= 0; i--) {
		while (!stack.isEmpty() && stack.peek() < nums2[i]) {
			stack.pop();
		}

		nextGreater.put(nums2[i], stack.isEmpty() ? -1 : stack.peek());
		stack.push(nums2[i]);
	}

	for (int i = 0; i < nums1.length; i++) {
		result[i] = nextGreater.get(nums1[i]);
	}

	return result;
}

public int[] nextGreaterElements(int[] nums) {
	int n = nums.length;
	int[] result = new int[n];
	Stack<Integer> stack = new Stack<>();
	int i = n - 1;
	int j = 1;

	while (i >= 0 && j <= 2) {
		while (!stack.isEmpty() && nums[i] >= stack.peek()) {
			stack.pop();
		}

		result[i] = stack.isEmpty() ? -1 : stack.peek();
		stack.push(nums[i]);
		i--;

		if (i == -1) {
			i = n - 1;
			j++;
		}
	}

	return result;
}


169. Majority Element

Boyer-Moore Majority Voting Algorithm
1. Initialize variable candidate to the first element of the array and a count variable to 1.
2. Iterate the array from 2nd element and if the current element is same as the majority element then increment the count else decrement it.
3. If the count becomes 0 then update the candidate to the current element and reset count to 1.
4. In the end again iterate the array and find the count of candidate element. If it is greater than 50% of elements then return it, else return -1.

public int majorityElement(int[] nums) {
	int n = nums.length;
	
	if (n == 1) return nums[0];

	int candidate = nums[0];
	int count = 1;

	for (int i = 1; i < n; i++) {
		if (nums[i] == candidate) {
			count++;
		} else {
			count--;

			if (count == 0) {
				candidate = nums[i];
				count++;
			}
		}
	}

	count = 0;
	for (int i = 0; i < n; i++) {
		if (nums[i] == candidate) {
			count++;
		}
	}

	if (count > (n / 2)) return candidate;

	return -1;
}

Approach:

2. Iterate the Set and check if the element 'e' is the start of the sequence i.e there should not exist element 'e' - 1 in the Set.
3. Once you find the element which is the start of the sequence, then keep incrementing that element and check if it exist in the Set. If it does, then update the length which is the max of current length and current element in sequence minus the start element of the sequence.

public int longestConsecutive(int[] nums) {
	int n = nums.length;

	if (n == 0) return 0;

	Set<Integer> numsSet = new HashSet();
	int maxLength = 1;
	
	for (int i = 0; i < n; i++) {
		numsSet.add(nums[i]);
	}

	for (int x: numsSet) {
		if (!numsSet.contains(x - 1)) {
			int y = x + 1;

			while (numsSet.contains(y)) {
				maxLength = Math.max(maxLength, y - x + 1);
				y++;
			}
		}
	}

	return maxLength;
}

public List<List<String>> groupAnagrams(String[] strs) {
	Map<String, List<String>> groupedAnagrams = new HashMap<>();

	for (String str: strs) {
		int[] count = new int[26];

		for (char c: str.toCharArray()) {
			count[c - 'a']++;
		}

		StringBuilder sb = new StringBuilder();

		for (int i = 0; i < 26; i++) {
			sb.append(count[i]).append("#");
		}

		String key = sb.toString();

		if (groupedAnagrams.containsKey(key)) {
			groupedAnagrams.get(key).add(str);
		} else {
			groupedAnagrams.put(key, 
			new ArrayList<>(Arrays.asList(str)));
		}
	}

	return new ArrayList<>(groupedAnagrams.values());
}


42. Trapping Rain Water

Approach:
1. Find the highest bar. You have to find how much water is trapped to the left and right of this bar because water cannot get trapped on the highest bar.
2. On the left and right side find the highest bar. The water trapped at each point/bar will the left/right highest bar minus current bar.

Code:
public int trap(int[] height) {
	int n = height.length;
	int sum = 0;
	int highest = height[0];
	int highestIdx = 0;

	for (int i = 0; i < n; i++) {
		if (height[i] > highest) {
			highest = height[i];
			highestIdx = i;
		}
	}

	int leftHighest = height[0];

	for (int i = 0; i < highestIdx; i++) {
		leftHighest = Math.max(leftHighest, height[i]);

		sum += leftHighest - height[i];
	}

	int rightHighest = height[n - 1];

	for (int i = n - 1; i > highestIdx; i--) {
		rightHighest = Math.max(rightHighest, height[i]);

		sum += rightHighest - height[i];
	}

	return sum;
}

560. Subarray Sum Equals K

Approach:
1. We maintain a Map which tracks the sum of elements upto i and its count for all elements i in the given array.
2. While iterating elements from 0 to n, find the sum of all elements upto element i.
3. Then check if (sum - k) exists in the Map.

Code:
public int subarraySum(int[] nums, int k) {
	int n = nums.length;
	Map<Integer, Integer> prefixSumMap = new HashMap<>();
	prefixSumMap.put(0, 1);
	int sum = 0;
	int count = 0;

	for (int i = 0; i < n; i++) {
		sum += nums[i];

		if (prefixSumMap.containsKey((int) (sum - k))) {
			count += prefixSumMap.get((int) (sum - k));
		}

		prefixSumMap.put(sum, prefixSumMap.getOrDefault(sum, 0) + 1);
	}

	return count;
}

438. Find All Anagrams in a String

Approach:
1. Create an array of size 26 (26 alphabets in English) and increment the count of array elements corrosponding to the alphabets of p. In the same loop do the same with alphabets of s, so that the time complexity is truly O(n). Basically we are finding the frequency of characters of the given string.
2. Then compare the array of initial window.
3. Then iterate the string 's' from window size i.e. length of string 'p' because we just compared the arrays upto the window size.
4. Keep updating the frequency of characters. Increment towards the right and decrement towards the left and check if both arrays match.


Code:
public List<Integer> findAnagrams(String s, String p) {
	int sLength = s.length();
	int pLength = p.length();
	List<Integer> anagramIndices = new ArrayList<>();

	if (pLength > sLength) return anagramIndices;

	int[] sCount = new int[26];
	int[] pCount = new int[26];

	for (int i = 0; i < pLength; i++) {
		pCount[p.charAt(i) - 'a']++;
		sCount[s.charAt(i) - 'a']++;
	}

	if (Arrays.equals(sCount, pCount)) {
		anagramIndices.add(0);
	}

	for (int i = pLength; i < sLength; i++) {
		sCount[s.charAt(i) - 'a']++;

		sCount[s.charAt(i - pLength) - 'a']--;

		if (Arrays.equals(sCount, pCount)) {
			anagramIndices.add(i - pLength + 1);
		}
	}

	return anagramIndices;
}

public boolean checkInclusion(String s1, String s2) {
	int s1Length = s1.length();
	int s2Length = s2.length();

	if (s1Length > s2Length) {
		return false;
	}

	int[] s1Count = new int[26];
	int[] s2Count = new int[26];

	for (int i = 0; i < s1Length; i++) {
		s1Count[s1.charAt(i) - 'a']++;
		s2Count[s2.charAt(i) - 'a']++;
	}

	if (Arrays.equals(s1Count, s2Count)) {
		return true;
	}

	for (int i = s1Length; i < s2Length; i++) {
		s2Count[s2.charAt(i) - 'a']++;

		s2Count[s2.charAt(i - s1Length) - 'a']--;

		if (Arrays.equals(s1Count, s2Count)) {
			return true;
		}
	}

	return false;
}

3. Longest Substring Without Repeating Characters

public int lengthOfLongestSubstring(String s) {
	Set<Character> charSet = new HashSet<>();
	int maxLength = 0;
	int left = 0, right = 0;

	while (right < s.length()) {
		char c = s.charAt(right);

		while (charSet.contains(c) && left < right) {
			charSet.remove(s.charAt(left));
			left++;
		}

		charSet.add(c);
		maxLength = Math.max(maxLength, right - left + 1);
		right++;
	}

	return maxLength;
}

54. Spiral Matrix

public List<Integer> spiralOrder(int[][] matrix) {
	int rowLen = matrix[0].length;
	int colLen = matrix.length;

	int left = 0, right = rowLen - 1, top = 0, bottom = colLen - 1;

	List<Integer> result = new ArrayList<>();

	while (left <= right && top <= bottom) {
		for (int i = left; i <= right; i++) {
			result.add(matrix[top][i]);
		}
		top++;

		for (int i = top; i <= bottom; i++) {
			result.add(matrix[i][right]);
		}
		right--;

		if (top <= bottom) {
			for (int i = right; i >= left; i--) {
				result.add(matrix[bottom][i]);
			}
			bottom--;
		}

		if (left <= right) {
			for (int i = bottom; i >= top; i--) {
				result.add(matrix[i][left]);
			}
			left++;
		}
	}

	return result;
}

48. Rotate Image

public void rotate(int[][] matrix) {
	int rowLength = matrix.length;
	int colLength = matrix[0].length;

	// transpose the matrix
	for (int i = 0; i < rowLength; i++) {
		for (int j = i + 1; j < colLength; j++) {
			int t = matrix[i][j];
			matrix[i][j] = matrix[j][i];
			matrix[j][i] = t;
		}
	}

	// reverse the numbers in each row
	for (int i = 0; i < rowLength; i++) {
		int left = 0, right = colLength - 1;

		while (left < right) {
			int t = matrix[i][left];
			matrix[i][left] = matrix[i][right];
			matrix[i][right] = t;
			left++;
			right--;
		}
	}
}

19. Remove Nth Node From End of List

Approach:
1. Maintain a slow and fast pointer.
2. Move the fast pointer by n nodes so that when the fast pointer reaches the last node, the slow pointer will reach the node which is one node behind the node to be removed.
3. If the fast pointer has become null after moving it n steps, then the node to be removed was the first node. Since 1 >= n <= sz. sz is the size of linked list.
Basically if n = sz, then the node to be removed is the head node.

Code:
public ListNode removeNthFromEnd(ListNode head, int n) {
	ListNode slow = head;
	ListNode fast = head;

	for (int i = 0; i < n; i++) {
		if (fast != null) {
			fast = fast.next;
		}
	}

	if (fast == null) {
		return slow.next;
	}

	while (fast.next != null) {
		slow = slow.next;
		fast = fast.next;
	}

	slow.next = slow.next.next;

	return head;
}

647. Palindromic Substring

class Solution {
    public int countSubstrings(String s) {
        int count = 0;

        for (int i = 0; i < s.length(); i++) {
            // For odd length palindromes
            count += findCount(s, i, i);

            // For even length palindromes
            count += findCount(s, i, i + 1);
        }

        return count;
    }

    private int findCount(String s, int left, int right) {
        int count = 0;

        while (left >= 0 && right < s.length() && s.charAt(left) == s.charAt(right)) {
            count++;
            left--;
            right++;
        }

        return count;
    }
}
